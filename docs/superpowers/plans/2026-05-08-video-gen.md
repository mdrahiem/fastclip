# Video generator (LinkedIn post → Remotion MP4) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship an anonymous Next.js wizard that accepts pasted English post text, calls an LLM to produce a versioned `SlidePlan` JSON, renders one MP4 per job (9:16 or 16:9) via Remotion on a single Node host, and supports built-in or uploaded background music.

**Architecture:** A pnpm monorepo isolates **contracts** (Zod + template/theme constants), **planner** (OpenAI structured JSON → `SlidePlan`), and **Remotion** (compositions reading props). The **web app** owns SQLite job state, a **single-concurrency in-process worker loop** that runs plan → normalize audio (ffmpeg) → `bundle` + `renderMedia`, writes files under `./data/jobs/<id>/`, exposes **poll + download** routes, and schedules deletion after **1 hour** (configurable).

**Tech Stack:** pnpm workspaces, TypeScript 5.x, Next.js 15 (App Router, **Node runtime** on the deployment target—not serverless), React 19, Tailwind CSS 4 (or 3—pick one and keep consistent), Zod, Drizzle ORM + **better-sqlite3**, OpenAI official SDK (`openai`), Remotion 4.x (`@remotion/bundler`, `@remotion/renderer`), ffmpeg via **`@ffmpeg-installer/ffmpeg`** + **`fluent-ffmpeg`**, Vitest for packages.

**Deployment note:** Long-running renders require a **container/VM** (Docker, Fly.io, Railway, EC2, etc.). Do not deploy the render path to Vercel Serverless as-is.

```typescript
import { z } from "zod";

export const TextLayerSchema = z.object({
  type: z.literal("text"),
  role: z.enum(["hook", "body", "cta", "label"]),
  text: z.string().min(1).max(320),
  region: z.enum(["top", "center", "bottom"]),
});

export const ShapeLayerSchema = z.object({
  type: z.literal("shape"),
  shape: z.enum(["circle", "rect", "line"]),
  region: z.enum(["top", "center", "bottom"]),
  accent: z.boolean().optional(),
});

export const LayerSchema = z.discriminatedUnion("type", [
  TextLayerSchema,
  ShapeLayerSchema,
]);

export const SlideSchema = z.object({
  index: z.number().int().min(0),
  layers: z.array(LayerSchema).min(1).max(8),
});

export const SlidePlanSchema = z.object({
  slidePlanVersion: z.literal(1),
  meta: z
    .object({
      title: z.string().max(120).optional(),
    })
    .optional(),
  slides: z.array(SlideSchema).min(1),
});

export type SlidePlan = z.infer<typeof SlidePlanSchema>;
export type Slide = z.infer<typeof SlideSchema>;
export type Layer = z.infer<typeof LayerSchema>;
```

Create `packages/contracts/src/validate-slide-plan.ts`:

```typescript
import { SlidePlanSchema, type SlidePlan } from "./slide-plan.js";

export class SlidePlanValidationError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "SlidePlanValidationError";
  }
}

export function parseSlidePlanJson(raw: string): SlidePlan {
  let parsed: unknown;
  try {
    parsed = JSON.parse(raw);
  } catch {
    throw new SlidePlanValidationError("Slide plan JSON could not be parsed.");
  }

  const result = SlidePlanSchema.safeParse(parsed);
  if (!result.success) {
    throw new SlidePlanValidationError("Slide plan failed schema validation.");
  }

  return result.data;
}

export function assertSlideCount(plan: SlidePlan, expected: number): void {
  if (plan.slides.length !== expected) {
    throw new SlidePlanValidationError(
      `Expected ${expected} slides, received ${plan.slides.length}.`,
    );
  }

  for (let i = 0; i < plan.slides.length; i += 1) {
    if (plan.slides[i]?.index !== i) {
      throw new SlidePlanValidationError(`Slide index mismatch at position ${i}.`);
    }
  }
}
```

Create `packages/planner/src/plan-post.ts`:

```typescript
import {
  assertSlideCount,
  getTemplateById,
  parseSlidePlanJson,
  SlidePlanValidationError,
  type SlidePlan,
  type VideoTemplate,
} from "@video-gen/contracts";
import OpenAI from "openai";
import { buildPlannerSystemPrompt, buildPlannerUserPrompt } from "./prompt.js";

export class PlannerError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "PlannerError";
  }
}

export type PlanPostInput = {
  openaiApiKey: string;
  model: string;
  templateId: VideoTemplate["id"];
  postText: string;
};

export function finalizePlanFromModelJson(
  raw: string,
  templateId: VideoTemplate["id"],
): SlidePlan {
  try {
    const template = getTemplateById(templateId);
    const plan = parseSlidePlanJson(raw);
    assertSlideCount(plan, template.slideCount);
    return plan;
  } catch (err) {
    const message =
      err instanceof SlidePlanValidationError || err instanceof Error
        ? err.message
        : "Slide plan validation failed.";
    throw new PlannerError(message);
  }
}

export async function planPost(input: PlanPostInput): Promise<SlidePlan> {
  const client = new OpenAI({ apiKey: input.openaiApiKey });
  const template = getTemplateById(input.templateId);
  const system = buildPlannerSystemPrompt(template);
  const user = buildPlannerUserPrompt(input.postText);

  let raw = "";
  try {
    const response = await client.chat.completions.create({
      model: input.model,
      temperature: 0.4,
      messages: [
        { role: "system", content: system },
        { role: "user", content: user },
      ],
    });

    raw = response.choices[0]?.message?.content ?? "";
    if (!raw) {
      throw new PlannerError("OpenAI returned an empty response.");
    }
  } catch (err) {
    if (err instanceof PlannerError) throw err;
    throw new PlannerError(
      err instanceof Error ? err.message : "OpenAI request failed.",
    );
  }

  return finalizePlanFromModelJson(raw, input.templateId);
}
```

Create `apps/web/lib/remotion/render.ts`:

```typescript
import path from "node:path";
import { bundle } from "@remotion/bundler";
import { renderMedia, selectComposition } from "@remotion/renderer";
import type { AspectRatioId, SlidePlan, ThemePack } from "@video-gen/contracts";

export type RenderRequest = {
  remotionEntry: string; // absolute path to remotion/src/index.ts
  outputLocation: string; // absolute mp4 path
  slidePlan: SlidePlan;
  slideDurationsSec: number[];
  theme: ThemePack;
  aspectRatioId: AspectRatioId;
  audioSrc: string;
};

const fps = 30;

function dimensionsForAspect(aspect: AspectRatioId): { width: number; height: number } {
  if (aspect === "9:16") return { width: 1080, height: 1920 };
  return { width: 1920, height: 1080 };
}

export async function renderLinkedInPostVideo(req: RenderRequest): Promise<void> {
  const { width, height } = dimensionsForAspect(req.aspectRatioId);

  const entryPoint = req.remotionEntry;
  const serveUrl = await bundle({
    entryPoint,
    webpackOverride: (config) => config,
  });

  const compositionId = "PostVideo";

  const durationInFrames =
    req.slideDurationsSec.reduce((sum, s) => sum + s, 0) * fps;

  const inputProps = {
    slidePlan: req.slidePlan,
    slideDurationsSec: req.slideDurationsSec,
    theme: req.theme,
    aspectRatioId: req.aspectRatioId,
    audioSrc: req.audioSrc,
  };

  const composition = await selectComposition({
    serveUrl,
    id: compositionId,
    inputProps,
  });

  await renderMedia({
    composition: {
      ...composition,
      width,
      height,
      durationInFrames,
      fps,
    },
    serveUrl,
    codec: "h264",
    outputLocation: req.outputLocation,
    inputProps,
    ffmpegOverride: ({ ffmpegOptions }) => ({
      ...ffmpegOptions,
      "-pix_fmt": "yuv420p",
    }),
  });
}
```

**Note:** `selectComposition` + overriding `durationInFrames`/`width`/`height` must stay consistent with what `PostVideo` reads from metadata. Preferred: compute duration only in `calculateMetadata` inside Remotion using inputProps; then **do not** override `durationInFrames` here. Implementation task: align `PostVideo` metadata and this render helper in one commit.

```typescript
import ffmpegInstaller from "@ffmpeg-installer/ffmpeg";
import ffmpeg from "fluent-ffmpeg";
import { mkdir } from "node:fs/promises";
import path from "node:path";

ffmpeg.setFfmpegPath(ffmpegInstaller.path);

export async function normalizeMusicToAac(inputPath: string, outputPath: string): Promise<void> {
  await mkdir(path.dirname(outputPath), { recursive: true });

  await new Promise<void>((resolve, reject) => {
    ffmpeg(inputPath)
      .audioCodec("aac")
      .audioBitrate("192k")
      .outputOptions(["-ar 48000", "-ac 2"])
      .on("end", () => resolve())
      .on("error", (err) => reject(err))
      .save(outputPath);
  });
}
```
